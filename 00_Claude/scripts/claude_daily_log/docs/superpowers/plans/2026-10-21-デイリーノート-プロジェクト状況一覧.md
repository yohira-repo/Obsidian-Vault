**Goal:** `/daily-sync` のたびに、Obsidian デイリーノートへ alpha 系・coop 系の「プロジェクト状況」表（ブランチ / 未コミット / 最終コミット）を自動更新する。

## Global Constraints

- 外部ライブラリを追加しない（既存方針どおり標準ライブラリのみ）。
- 判定条件と表記の**正本は Python 版 `status.py`**。alphasystem `.claude/hooks/project-status.sh` と同一に保ち、両ファイル先頭コメントに整合ルールを明記する。対象: 列の並び、`⚠upstream消失` と `(未push)` の判定、未コミットの `N件`、相対時刻の日本語化、件名 40 文字切り詰め、`|` エスケープ。
- 列の並びは固定: `プロジェクト | ブランチ | 未コミット | 最終コミット | メッセージ`。
- **行の並びは「各グループの先頭が傘プロジェクト（alphasystem / coop）であること」のみが要件。以降の順序は任意**（ユーザー決定 2026-10-21）。設計書 3 章の「構成表の順」より緩い条件のため、`list_projects()` の既存の名前昇順をそのまま使う。
- マーカーは `<!-- project-status:start -->` / `<!-- project-status:end -->`。見出しは `## プロジェクト状況`。
- 位置は `## Claude作業ログ` の**直前**。初回はその直前に挿入、作業ログ見出しが無ければ末尾。以後はマーカー位置を尊重する。
- プロジェクト名リンクは `[[00_Claude/projects/<name>|<name>]]`。同居行（alphabsmail）は**リンクを付けず**ブランチ列に `alphasystem内`、他列は `-`。
- 表の下に `（YYYY-MM-DD HH:MM 時点）` を 1 行。
- **時刻行は「表の中身が変わったときだけ」更新する**（ユーザー決定 2026-10-21）。表の中身（ブランチ / 未コミット / 最終コミット / メッセージ）が前回と同一なら、時刻行も含めてブロックを書き換えず「変更なし」とする。時刻行をマーカー内に置いたまま毎回更新すると、分が変わるたびにノートが書き換わって同期のたびに git 差分が出て、仕様 5 章の「変更なし」が成立しなくなるため。この結果、時刻は「最後に状況が変わった時刻」を意味する。
- 件名の切り詰めは **40 文字**（`MSG_MAX = 40`）、超過時は末尾に `…`。
- 全体同期（`sync` / `auto`）でのみ更新する。`--project` 単体同期では**更新しない**。
- 状況収集が失敗しても作業ログの更新を止めない。状況一覧は前回のまま残し、報告に 1 行加える。
- デイリーノートが未作成の日はノートを作らない（既存方針を踏襲）。
- 既存テスト 147 件を壊さないこと（着手前のベースライン: `Ran 147 tests ... OK`）。

"""CLAUDE.md の構成テーブルから対象プロジェクト一覧を得る。"""
import os
import re
from dataclasses import dataclass
from typing import List, Sequence, Tuple

ROW_RE = re.compile(r"^\s*\|\s*([A-Za-z0-9_.-]+)\s*\|\s*(\.{1,2}/[^|]+?)\s*\|")

DEFAULT_CONFIG_FILES = ("alphasystem/CLAUDE.md", "coop/CLAUDE.md")

KIND_REPO = "repo"
KIND_NESTED = "nested"


@dataclass(frozen=True)
class Project:
    name: str
    path: str
    kind: str = KIND_REPO

    @property
    def is_nested(self) -> bool:
        """構成表で `./` 配下に置かれた同居ディレクトリ（独立リポジトリではない）。"""
        return self.kind == KIND_NESTED

    @property
    def exists(self) -> bool:
        if self.is_nested:
            # 同居は .git を持たない。ディレクトリの有無で判定する。
            return os.path.isdir(self.path)
        return os.path.isdir(os.path.join(self.path, ".git"))


def parse_project_table(text: str) -> List[Tuple[str, str]]:
    """構成表の行を (名前, 相対パス) で返す。

    `../` 行は独立リポジトリ、`./` 行は同居ディレクトリ。重複名は初出を優先する。
    """
    rows: List[Tuple[str, str]] = []
    seen = set()
    for line in text.splitlines():
        matched = ROW_RE.match(line)
        if not matched:
            continue
        name = matched.group(1)
        if name in seen:
            continue
        seen.add(name)
        rows.append((name, matched.group(2)))
    return rows


def list_projects(git_root: str, config_files: Sequence[str] = DEFAULT_CONFIG_FILES) -> List[Project]:
    """プロジェクト一覧（名前昇順）。重複名は初出優先。

    相対パスは、その CLAUDE.md が置かれたリポジトリのディレクトリ基準で解決する
    （`./alphabsmail` は `<git_root>/alphasystem/alphabsmail`）。

    並び順は呼び出し側の要件ではない（状況一覧は描画側で傘プロジェクトを先頭に
    並べ替える）。従来どおり名前昇順で返す。
    """
    found: List[Project] = []
    seen = set()
    for relative in config_files:
        config_path = os.path.join(git_root, relative)
        owner_dir = os.path.dirname(config_path)
        try:
            with open(config_path, encoding="utf-8") as handle:
                text = handle.read()
        except OSError:
            continue
        for name, rel_path in parse_project_table(text):
            if name in seen:
                continue
            seen.add(name)
            kind = KIND_NESTED if rel_path.startswith("./") else KIND_REPO
            path = os.path.normpath(os.path.join(owner_dir, rel_path))
            found.append(Project(name=name, path=path, kind=kind))
    return sorted(found, key=lambda project: project.name)
```

【整合ルール】判定条件と表記（列の並び、⚠upstream消失 / (未push) の判定、
未コミットの「N件」、相対時刻の日本語化、件名の40文字切り詰め、| のエスケープ）は
alphasystem の .claude/hooks/project-status.sh（セッション内表示用の bash 版）と
同一に保つ。**正本はこの Python 版**。どちらかを変えるときは必ずもう一方にも

import os
import re
import subprocess
from dataclasses import dataclass
from typing import List, Optional

MSG_MAX = 40
GIT_TIMEOUT = 20

NOT_CLONED = "未clone"
NOT_A_REPO = "（git管理外）"
NESTED_LABEL = "alphasystem内"
FAILED_LABEL = "（取得失敗）"
DASH = "-"

# bash 版の sed と同じ順序・同じ置換。順序を変えると複合表記が崩れる。
_RELATIVE_RULES = (
    (re.compile(r" seconds? ago"), "秒前"),
    (re.compile(r" minutes? ago"), "分前"),
    (re.compile(r" hours? ago"), "時間前"),
    (re.compile(r" days? ago"), "日前"),
    (re.compile(r" weeks? ago"), "週間前"),
    (re.compile(r" months? ago"), "か月前"),
    (re.compile(r" years? ago"), "年前"),
    (re.compile(r" years?,"), "年"),
    (re.compile(r" months?$"), "か月"),
    (re.compile(r", "), "・"),
    (re.compile(r"年 "), "年"),
)


@dataclass(frozen=True)
class ProjectStatus:
    name: str
    branch: str
    dirty: str
    when: str
    message: str
    linked: bool
    failed: bool


def japanize_relative(text: str) -> str:
    """git の `%cr`（英語）を日本語表記にする。未知の表記はそのまま返す。"""
    result = text
    for pattern, replacement in _RELATIVE_RULES:
        result = pattern.sub(replacement, result)
    return result


def truncate_subject(text: str, limit: int = MSG_MAX) -> str:
    """件名を limit 文字で切り詰め、超過時は末尾に `…` を付ける。"""
    if len(text) > limit:
        return text[:limit] + "…"
    return text


def escape_cell(text: str) -> str:
    """表のセルに入れるため `|` をエスケープする。"""
    return text.replace("|", r"\|")


def _git(path: str, *args: str) -> Optional[str]:
    """git を実行して標準出力を返す。失敗・異常終了は None。"""
    try:
        proc = subprocess.run(
            ["git", "-C", path] + list(args),
            capture_output=True, timeout=GIT_TIMEOUT,
        )
    except (OSError, subprocess.SubprocessError):
        return None
    if proc.returncode != 0:
        return None
    return proc.stdout.decode("utf-8", "replace").strip()


def _plain(name: str, branch: str, linked: bool = True, failed: bool = False) -> ProjectStatus:
    """ブランチ列だけに状態を出し、残りを `-` で埋めた行。"""
    return ProjectStatus(
        name=name, branch=branch, dirty=DASH, when=DASH, message=DASH,
        linked=linked, failed=failed,
    )


def _branch_label(path: str) -> str:
    """ブランチ名に `⚠upstream消失` / `(未push)` のフラグを付けた表示。"""
    branch = _git(path, "branch", "--show-current")
    if branch is None:
        return FAILED_LABEL
    if not branch:
        short = _git(path, "rev-parse", "--short", "HEAD") or "?"
        return "(detached %s)" % short

    remote = _git(path, "config", "branch.%s.remote" % branch) or ""
    merge = _git(path, "config", "branch.%s.merge" % branch) or ""
    if remote and merge:
        ref = "refs/remotes/%s/%s" % (remote, merge[len("refs/heads/"):] if merge.startswith("refs/heads/") else merge)
        if _git(path, "show-ref", "--verify", "--quiet", ref) is None:
            return branch + " ⚠upstream消失"
    elif branch not in ("main", "staging"):
        return branch + " (未push)"
    return branch


def collect_status(project) -> ProjectStatus:
    """1 プロジェクト分の状況。例外は投げず、必ず ProjectStatus を返す。"""
    name = project.name
    path = project.path

    if project.is_nested:
        # 同居ディレクトリ: 独立リポジトリではないのでリンクも git 情報も出さない。
        return _plain(name, NESTED_LABEL, linked=False)

    if not os.path.isdir(path):
        return _plain(name, NOT_CLONED)

    if _git(path, "rev-parse", "--is-inside-work-tree") is None:
        return _plain(name, NOT_A_REPO)

    try:
        branch = _branch_label(path)

        porcelain = _git(path, "status", "--porcelain")
        if porcelain is None:
            dirty = DASH
        else:
            count = len([line for line in porcelain.splitlines() if line.strip()])
            dirty = "%d件" % count if count else DASH

        raw_when = _git(path, "log", "-1", "--format=%cr")
        when = japanize_relative(raw_when) if raw_when else DASH

        raw_message = _git(path, "log", "-1", "--format=%s")
        message = escape_cell(truncate_subject(raw_message)) if raw_message else DASH
    except Exception:
        # 1 リポジトリの失敗で表全体を止めない。
        return _plain(name, FAILED_LABEL, failed=True)

    return ProjectStatus(
        name=name, branch=branch, dirty=dirty, when=when, message=message,
        linked=True, failed=branch == FAILED_LABEL,
    )


def collect_all(projects: List) -> List[ProjectStatus]:
    """与えられた順序を保ったまま全プロジェクトの状況を集める。"""
    return [collect_status(project) for project in projects]

    """1 プロジェクト分の表の行。同居はリンクを付けない。"""
    label = (
        "[[00_Claude/projects/%s|%s]]" % (item.name, item.name)
        if item.linked else item.name
    )
    return "| %s | %s | %s | %s | %s |" % (

    )


def _umbrella_first(group_items: List, umbrella: Optional[str]) -> List:
    """傘プロジェクトだけを先頭へ移す。それ以外の順序は入力のまま保つ。

    並び順の要件は「各グループの先頭が傘（alphasystem / coop）であること」だけで、
    以降の順序は問わない（2026-10-21 のユーザー判断）。
    """
    if umbrella is None:
        return list(group_items)
    head = [item for item in group_items if item.name == umbrella]
    rest = [item for item in group_items if item.name != umbrella]
    return head + rest


def render_status_lines(statuses: List, now_text: str) -> List[str]:
    """`### alpha` / `### coop` の表 2 つと時刻行を描画する。

    各グループの先頭に傘プロジェクト（alphasystem / coop）を置き、以降は
    statuses の順序をそのまま使う。該当が 1 件も無いグループは丸ごと省略する。
    「その他」グループは状況一覧には出さない（仕様 3 章の表は alpha 系・coop 系の 2 つだけ）。

    now_text は末尾の時刻行にだけ使う。表の中身が前回と同じなら呼び出し側
    （_replace_block）がブロックごと据え置くため、時刻も更新されない。
    """
    buckets: Dict[int, List] = {0: [], 1: []}
    for item in statuses:
        index = _group_index(item.name)
        if index in buckets:
            buckets[index].append(item)

    lines: List[str] = []
    for group in (0, 1):
        group_items = _umbrella_first(buckets[group], _GROUP_UMBRELLAS[group])
        if not group_items:
            continue
        if lines:
            lines.append("")
        lines.append(STATUS_GROUP_HEADINGS[group])
        lines.append("")
        lines.extend(STATUS_TABLE_HEADER)
        lines.extend(_status_row(item) for item in group_items)

    if lines:
        lines.append("")
    lines.append("（%s 時点）" % now_text)
    return lines

    volatile_prefix が None なら単純一致。指定時は、その文字列で始まる行を
    両者から取り除いてから比較する。表の中身が同じなら時刻だけのために
    ノートを書き換えないための判定（2026-10-21 のユーザー決定）。
    """
    if volatile_prefix is None:
        return current == block
    def strip(seq):
        return [line for line in seq if not line.startswith(volatile_prefix)]
    return strip(current) == strip(block)


def _replace_block(lines, start_marker, end_marker, body, heading, anchor_heading, label,
                   volatile_prefix=None):
    """マーカー区間を body で置き換える。無ければ挿入する。

    戻り値は (新しい行リスト, 変化したか, 警告 or None)。マーカーが壊れている
    場合は何も変更せず警告を返す（他方のブロックの更新は止めない）。
    volatile_prefix を渡すと、その行を無視して同値判定する（時刻行用）。
    """
    start_indices = _marker_indices(lines, start_marker)
    end_indices = _marker_indices(lines, end_marker)
    start_count = len(start_indices)
    end_count = len(end_indices)

    block = [start_marker] + body + [end_marker]

    if start_count == 1 and end_count == 1:
        start_idx = start_indices[0]
        end_idx = end_indices[0]
        if end_idx < start_idx:
            return lines, False, _marker_error_message(label, start_count, end_count, order_wrong=True)
        current = lines[start_idx:end_idx + 1]
        if _same_ignoring_volatile(current, block, volatile_prefix):
            # 中身が同じなら時刻行も据え置く（毎分の書き換えと git 差分を避ける）。
            return lines, False, None
        return lines[:start_idx] + block + lines[end_idx + 1:], True, None

    if start_count == 0 and end_count == 0:
        if anchor_heading is not None:
            for index, line in enumerate(lines):
                if line.strip() == anchor_heading:
                    inserted = [heading, ""] + block + [""]
                    return lines[:index] + inserted + lines[index:], True, None
        trimmed = list(lines)
        while trimmed and trimmed[-1].strip() == "":
            trimmed.pop()
        prefix = trimmed + [""] if trimmed else []
        return prefix + [heading, ""] + block, True, None

    return lines, False, _marker_error_message(label, start_count, end_count, order_wrong=False)


def update_daily(
    vault: str,
    date: str,
    latest_entries: Dict[str, object],
    project_names: Optional[List[str]] = None,
    status_lines: Optional[List[str]] = None,
) -> Tuple[bool, bool, List[str]]:
    """管理ブロック（作業ログ / プロジェクト状況）を再生成する。

    戻り値は (作業ログが変わったか, 状況一覧が変わったか, 警告一覧)。
    `status_lines` が None のときは状況一覧に一切触れない。

    Daily ノートが存在しない場合は**何も作らない**（利用者が毎朝テンプレートから
    作成する運用を壊さないため）。呼び出し側は事前に存在を確認して報告すること。
    """
    warnings: List[str] = []
    path = os.path.join(vault, daily_relpath(date))
    if not os.path.exists(path):
        return False, False, warnings

    with open(path, encoding="utf-8") as handle:
        original = handle.read()
    lines = original.splitlines()
    relpath = daily_relpath(date)

    # 作業ログ: マーカー破損は従来どおり例外にする（呼び出し側が警告へ変換する）。
    start_indices = _marker_indices(lines, START_MARKER)
    end_indices = _marker_indices(lines, END_MARKER)
    start_count = len(start_indices)
    end_count = len(end_indices)
    if start_count == 1 and end_count == 1 and end_indices[0] < start_indices[0]:
        raise DailyMarkerError(_marker_error_message(relpath, start_count, end_count, order_wrong=True))
    if (start_count, end_count) not in ((1, 1), (0, 0)):
        raise DailyMarkerError(_marker_error_message(relpath, start_count, end_count, order_wrong=False))

    block_body = render_project_lines(latest_entries, project_names or [])
    lines, log_changed, _ = _replace_block(
        lines, START_MARKER, END_MARKER, block_body,
        SECTION_HEADING, None, relpath,
    )

    status_changed = False
    if status_lines is not None:
        lines, status_changed, warning = _replace_block(
            lines, STATUS_START_MARKER, STATUS_END_MARKER, status_lines,
            STATUS_SECTION_HEADING, SECTION_HEADING,
            "%s のプロジェクト状況ブロック" % relpath,
            volatile_prefix=STATUS_TIME_PREFIX,
        )
        if warning:
            warnings.append(warning)

    new_text = "\n".join(lines).rstrip("\n") + "\n"
    if original == new_text:
        return False, False, warnings

    os.makedirs(os.path.dirname(path), exist_ok=True)
    atomicio.write_text_atomic(path, new_text)
    return log_changed, status_changed, warnings
```
